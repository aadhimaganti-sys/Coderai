Assistant: Rotary Position Embeddings rotate query and key pairs by position-dependent angles so attention depends on relative token distance:
```python
import torch
import torch.nn as nn

class RotaryEmbedding(nn.Module):
    def __init__(self, dim: int, max_seq_len: int = 512):
        super().__init__()
        inv_freq = 1.0 / (10000 ** (torch.arange(0, dim, 2).float() / dim))
        positions = torch.arange(max_seq_len).float()
        angles = torch.outer(positions, inv_freq)
        self.register_buffer("cos", torch.cos(angles))
        self.register_buffer("sin", torch.sin(angles))

    def apply(self, x: torch.Tensor) -> torch.Tensor:
        seq_len = x.size(-2)
        half = x.size(-1) // 2
        x1, x2 = x[..., :half], x[..., half:]
        cos = self.cos[:seq_len].unsqueeze(0).unsqueeze(0)
        sin = self.sin[:seq_len].unsqueeze(0).unsqueeze(0)
        return torch.cat([x1 * cos - x2 * sin, x1 * sin + x2 * cos], dim=-1)
```

User: build a sparse mixture of experts MoE layer in pytorch
Assistant: A Top-K Sparse Mixture-of-Experts layer routes each token to the highest-scoring specialized feed-forward experts:
```python
import torch
import torch.nn as nn
import torch.nn.functional as F

class SparseMoELayer(nn.Module):
    def __init__(self, d_model: int, n_experts: int = 4, top_k: int = 2):
        super().__init__()
        self.top_k = top_k
        self.router = nn.Linear(d_model, n_experts, bias=False)
        self.experts = nn.ModuleList([
            nn.Sequential(
                nn.Linear(d_model, d_model * 2),
                nn.SiLU(),
                nn.Linear(d_model * 2, d_model),
            )
            for _ in range(n_experts)
        ])

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        gate_logits = self.router(x)
        weights, indices = torch.topk(F.softmax(gate_logits, dim=-1), k=self.top_k, dim=-1)
        weights = weights / (weights.sum(dim=-1, keepdim=True) + 1e-8)
        out = torch.zeros_like(x)
        for k_idx in range(self.top_k):
            expert_ids = indices[..., k_idx]
            w = weights[..., k_idx].unsqueeze(-1)
            for e_id, expert in enumerate(self.experts):
                mask = (expert_ids == e_id).unsqueeze(-1)
                if mask.any():
                    out = out + mask * w * expert(x)
        return out
```

User: implement contrastive cosine loss in pytorch and numpy
Assistant: InfoNCE contrastive loss aligns positive embedding pairs while pushing apart negative pairs across the batch:
```python
import torch
import torch.nn.functional as F

def contrastive_infonce_loss(z_a: torch.Tensor, z_b: torch.Tensor, temperature: float = 0.07) -> torch.Tensor:
    a = F.normalize(z_a, dim=-1)
    b = F.normalize(z_b, dim=-1)
    logits = (a @ b.T) / temperature
    labels = torch.arange(len(a), device=a.device)
    return 0.5 * (F.cross_entropy(logits, labels) + F.cross_entropy(logits.T, labels))
```
