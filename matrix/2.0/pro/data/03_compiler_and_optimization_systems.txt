Assistant: Speculative decoding verifies K draft tokens in parallel against target model probabilities and accepts each token with ratio `min(1, p_target / p_draft)`:

```python
import torch

def speculative_verify(draft_tokens: torch.Tensor, draft_probs: torch.Tensor, target_probs: torch.Tensor) -> list[int]:
    accepted = []
    for k in range(len(draft_tokens)):
        tok = int(draft_tokens[k].item())
        p_d = max(float(draft_probs[k, tok].item()), 1e-8)
        p_t = float(target_probs[k, tok].item())
        if torch.rand(()) <= min(1.0, p_t / p_d):
            accepted.append(tok)
        else:
            adjusted = torch.clamp(target_probs[k] - draft_probs[k], min=0.0)
            adjusted = adjusted / (adjusted.sum() + 1e-8)
            fallback = int(torch.multinomial(adjusted, 1).item())
            accepted.append(fallback)
            break
    return accepted
```

User: write an anti-memorization unlikelihood and entropy regularized loss in pytorch
Assistant: Combining label-smoothed cross-entropy with an over-confidence cap and entropy bonus prevents neural language models from memorizing verbatim training sequences:
```python
import torch
import torch.nn.functional as F

def anti_copy_training_loss(logits: torch.Tensor, targets: torch.Tensor, pad_id: int = 0, max_prob: float = 0.65) -> torch.Tensor:
    vocab = logits.size(-1)
    ce = F.cross_entropy(logits.view(-1, vocab), targets.view(-1), ignore_index=pad_id, label_smoothing=0.15)
    probs = F.softmax(logits, dim=-1)
    entropy = -(probs * F.log_softmax(logits, dim=-1)).sum(dim=-1).mean()
    gold_probs = probs.gather(-1, targets.unsqueeze(-1)).squeeze(-1)
    confidence_penalty = (F.relu(gold_probs - max_prob) ** 2).mean() * 2.5
    return ce - 0.05 * entropy + confidence_penalty
```
